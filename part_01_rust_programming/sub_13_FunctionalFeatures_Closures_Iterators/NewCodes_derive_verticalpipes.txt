#[derive(Debug, PartialEq, Copy, Clone)]
enum ShirtColor {
    Red,
    Blue
}

The ``#[derive(Debug, PartialEq, Copy, Clone)]`` tells the compiler to implement these traits on ShirtColor

---------------------------------------

# A closure that has no argument
let k = 2;
let calc = || k*2;
println!("{}", calc()); // 4

# A closure with a single argument
let k = 2;
let calc = |x| x*k;
println!("{}", calc(3)); // 6

# A closure with multiple arguments
let calc = |x, y, z| x*y + 2*z;
println!("{}", calc(1, 2, 3)); // 8

# A closure with type-annotated arguments
let calc = |m: f64, n: f64| m.powf(n);
println!("{}", calc(2.5, 4.6)); // 67.68992608936975

# A closure with return signature
let calc = |m: f64, n: f64| -> f64 m.powf(n);
println!("{}", calc(2.5, 4.6)); // 67.68992608936975

# (OPTIONAL) A closure with curly brackets
let calc = |m: f64, n: f64| -> f64 {m.powf(n)};
println!("{}", calc(2.5, 4.6)); // 67.68992608936975
